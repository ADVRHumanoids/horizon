if(HORIZON_TEST_KYON_COLLISION)

    endif()
    if(NOT HORIZON_KYON_URDF OR NOT EXISTS "${HORIZON_KYON_URDF}")
        message(FATAL_ERROR "Set HORIZON_KYON_URDF to the canonical kyon.urdf path")

    add_executable(testKyonCollision testKyonCollision.cpp)
    target_link_libraries(testKyonCollision PRIVATE ilqr GTest::gtest_main pinocchio::pinocchio)
    target_compile_definitions(testKyonCollision PRIVATE HORIZON_KYON_URDF_PATH=\"${HORIZON_KYON_URDF}\")

    add_executable(profileKyonCollision profileKyonCollision.cpp)
    target_link_libraries(profileKyonCollision PRIVATE ilqr pinocchio::pinocchio)
    target_compile_definitions(profileKyonCollision PRIVATE HORIZON_KYON_URDF_PATH=\"${HORIZON_KYON_URDF}\")
    add_executable(benchmarkKyonTrajectory ../src/kyon_collision_example.cpp)
    target_link_libraries(benchmarkKyonTrajectory PRIVATE ilqr pinocchio::pinocchio)
    target_compile_definitions(benchmarkKyonTrajectory PRIVATE HORIZON_KYON_URDF_PATH=\"${HORIZON_KYON_URDF}\")

    set_tests_properties(kyonTrajectorySmoke PROPERTIES LABELS "robot;kyon;performance" TIMEOUT 120)
    if(HORIZON_WITH_KOKKOS)
